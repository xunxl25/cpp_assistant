# C++ 做题应用 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建本地 Streamlit 做题应用，支持按知识点刷题、错题本复习（准确率<75%）、AI 问答、题库管理。

**Architecture:** Python Streamlit 单体应用。题库用 JSON 文件存储，练习记录用 SQLite（practice_log 表，记录所有练过的题）。错题本是 practice_log 的查询视图。知识点展示/筛选用 question_id 从内存题库查（无 LIKE）。AI 问答嵌在答题界面，自动关联当前题。

**Tech Stack:** Python 3.10+、Streamlit、SQLite3（标准库）、python-dotenv、openai（兼容 DeepSeek 等）、pytest

**Spec:** `docs/superpowers/specs/2026-07-05-cpp-quiz-app-design.md`

**Git:** 项目当前非 git 仓库。Task 0 会初始化 git，后续每个 Task 末尾 commit。

```python
"""题库加载与筛选。只读 JSON，不碰数据库和 AI。"""
import json
import logging
from typing import Optional

logger = logging.getLogger(__name__)


def load_all_questions(bank_dir) -> list[dict]:
    """加载 bank_dir 下所有 *.json，返回题目列表。跳过解析失败的文件。"""
    bank_path = __import__("pathlib").Path(bank_dir)
    if not bank_path.exists():
        return []
    questions = []
    for f in sorted(bank_path.glob("*.json")):
        try:
            data = json.loads(f.read_text(encoding="utf-8"))
            if isinstance(data, list):
                questions.extend(data)
            else:
                logger.warning("%s 顶层不是数组，已跳过", f)
        except json.JSONDecodeError as e:
            logger.warning("解析 %s 失败：%s，已跳过", f, e)
    return questions


def get_by_id(questions: list[dict], question_id: str) -> Optional[dict]:
    """按 id 查找题目。"""
    for q in questions:
        if q.get("id") == question_id:
            return q
    return None


def filter_by_knowledge_point(questions: list[dict], kp: str) -> list[dict]:
    """筛选包含某知识点的题目。"""
    return [q for q in questions if kp in q.get("knowledge_points", [])]


def list_knowledge_points(questions: list[dict]) -> list[str]:
    """返回去重排序的知识点列表。"""
    s = set()
    for q in questions:
        for kp in q.get("knowledge_points", []):
            s.add(kp)
    return sorted(s)
```

  last_wrong_at     TEXT,
  last_attempt_at   TEXT,
  accuracy          REAL DEFAULT 0.0,
  mastered          INTEGER DEFAULT 0
);
"""


def _now_iso() -> str:
    return datetime.now().isoformat(timespec="seconds")


class PracticeTracker:
    def __init__(self, db_path):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._conn = sqlite3.connect(str(self.db_path))
        self._conn.row_factory = sqlite3.Row
        self._conn.executescript(SCHEMA)
        self._conn.commit()

    def close(self):
        self._conn.close()

    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.close()

    def get(self, question_id: str) -> Optional[dict]:
        row = self._conn.execute(
            "SELECT * FROM practice_log WHERE question_id = ?", (question_id,)
        ).fetchone()
        return dict(row) if row else None

    def record_answer(self, question_id: str, exam_type: str, exam_level: str,
                      is_correct: bool, user_answer: str):
        """记录一次作答。首次插入，否则更新。"""
        now = _now_iso()
        existing = self.get(question_id)
        if existing is None:
            wrong_count = 0 if is_correct else 1
            last_wrong_answer = None if is_correct else user_answer
            last_wrong_at = None if is_correct else now
            accuracy = 1.0 if is_correct else 0.0
            mastered = 1 if accuracy >= MASTERED_THRESHOLD else 0
            self._conn.execute(
                """INSERT INTO practice_log
                   (question_id, exam_type, exam_level, attempt_count, wrong_count,
                    last_wrong_answer, first_wrong_at, last_wrong_at, last_attempt_at,
                    accuracy, mastered)
                   VALUES (?, ?, ?, 1, ?, ?, ?, ?, ?, ?, ?)""",
                (question_id, exam_type, exam_level, wrong_count, last_wrong_answer,
                 now, last_wrong_at, now, accuracy, mastered)
            )
        else:
            attempt_count = existing["attempt_count"] + 1
            wrong_count = existing["wrong_count"] + (0 if is_correct else 1)
            last_wrong_answer = existing["last_wrong_answer"]
            last_wrong_at = existing["last_wrong_at"]
            if not is_correct:
                last_wrong_answer = user_answer
                last_wrong_at = now
            accuracy = (attempt_count - wrong_count) / attempt_count
            mastered = 1 if accuracy >= MASTERED_THRESHOLD else 0
            self._conn.execute(
                """UPDATE practice_log SET
                     attempt_count = ?, wrong_count = ?, last_wrong_answer = ?,
                     last_wrong_at = ?, last_attempt_at = ?, accuracy = ?, mastered = ?
                   WHERE question_id = ?""",
                (attempt_count, wrong_count, last_wrong_answer, last_wrong_at,
                 now, accuracy, mastered, question_id)
            )
        self._conn.commit()

    def mark_mastered(self, question_id: str):
        """手动移出错题本。"""
        self._conn.execute(
            "UPDATE practice_log SET mastered = 1 WHERE question_id = ?",
            (question_id,)
        )
        self._conn.commit()

    def get_wrong_view(self, order: str = "default") -> list[dict]:
        """错题本视图：wrong_count>0 且 mastered=0。"""
        order_clause = {
            "default": "last_wrong_at DESC",
            "high_freq": "wrong_count DESC, last_wrong_at DESC",
            "random": "RANDOM()",
        }.get(order, "last_wrong_at DESC")
        rows = self._conn.execute(
            f"SELECT * FROM practice_log WHERE wrong_count > 0 AND mastered = 0 ORDER BY {order_clause}"
        ).fetchall()
        return [dict(r) for r in rows]

    def get_all_practiced_ids(self) -> set[str]:
        rows = self._conn.execute("SELECT question_id FROM practice_log").fetchall()
        return {r["question_id"] for r in rows}

    def get_all(self) -> list[dict]:
        rows = self._conn.execute("SELECT * FROM practice_log").fetchall()
        return [dict(r) for r in rows]
```

"""大模型调用与上下文拼装。不碰 UI 和数据库。"""
import os
from pathlib import Path
from typing import Optional

try:
    from dotenv import load_dotenv
    load_dotenv(Path(__file__).resolve().parent.parent.parent / ".env")
except ImportError:
    pass


def build_context(question: dict, user_answer: str) -> str:
    """拼装发给大模型的题目上下文。"""
    lines = [f"题目：{question.get('stem', '')}", ""]
    options = question.get("options", {})
    if question.get("type") == "true_false":
        lines.append("选项：正确 / 错误")
    else:
        lines.append("选项：")
        for key, val in options.items():
            lines.append(f"  {key}. {val}")
    lines.append("")
    lines.append(f"用户答案：{user_answer}")
    lines.append(f"正确答案：{question.get('answer', '')}")
    explanation = question.get("explanation")
    if explanation:
        lines.append(f"参考解析：{explanation}")
    return "\n".join(lines)


def chat(user_question: str, question: dict, user_answer: str) -> str:
    """调用大模型。无 key 时返回提示。"""
    api_key = os.environ.get("LLM_API_KEY", "")
    if not api_key:
        return "未配置 LLM_API_KEY，请在项目根目录 .env 中填写（参考 .env.example）。"

    context = build_context(question, user_answer)
    prompt = f"{context}\n\n学生提问：{user_question}"

    try:
        from openai import OpenAI
        client = OpenAI(
            api_key=api_key,
            base_url=os.environ.get("LLM_BASE_URL") or None,
        )
        resp = client.chat.completions.create(
            model=os.environ.get("LLM_MODEL", "deepseek-chat"),
            messages=[
                {"role": "system", "content": "你是一位 C++ 编程辅导老师，用中文耐心解答学生的概念问题。"},
                {"role": "user", "content": prompt},
            ],
        )
        return resp.choices[0].message.content or ""
    except Exception as e:
        return f"调用大模型失败：{e}"


def chat_stream(user_question: str, question: dict, user_answer: str):
    """流式版本，逐 token 返回。无 key 时 yield 一条提示。"""
    api_key = os.environ.get("LLM_API_KEY", "")
    if not api_key:
        yield "未配置 LLM_API_KEY，请在项目根目录 .env 中填写（参考 .env.example）。"
        return

    context = build_context(question, user_answer)
    prompt = f"{context}\n\n学生提问：{user_question}"

    try:
        from openai import OpenAI
        client = OpenAI(
            api_key=api_key,
            base_url=os.environ.get("LLM_BASE_URL") or None,
        )
        stream = client.chat.completions.create(
            model=os.environ.get("LLM_MODEL", "deepseek-chat"),
            messages=[
                {"role": "system", "content": "你是一位 C++ 编程辅导老师，用中文耐心解答学生的概念问题。"},
                {"role": "user", "content": prompt},
            ],
            stream=True,
        )
        for chunk in stream:
            delta = chunk.choices[0].delta.content
            if delta:
                yield delta
    except Exception as e:
        yield f"调用大模型失败：{e}"
```

def render_knowledge_overview(questions, practiced_ids, tracker):
    """知识点概览柱状图：已练习/未练习 + 准确率标注。"""
    st.subheader("📊 知识点概览")
    kp_stats = {}
    for q in questions:
        for kp in q.get("knowledge_points", []):
            if kp not in kp_stats:
                kp_stats[kp] = {"total": 0, "practiced": 0, "correct_sum": 0.0, "practiced_ids": set()}
            kp_stats[kp]["total"] += 1
            if q["id"] in practiced_ids:
                kp_stats[kp]["practiced"] += 1
                kp_stats[kp]["practiced_ids"].add(q["id"])

    # 计算每个知识点已练习题的平均准确率
    for kp, s in kp_stats.items():
        if s["practiced_ids"]:
            accs = []
            for qid in s["practiced_ids"]:
                row = tracker.get(qid)
                if row:
                    accs.append(row["accuracy"])
            s["avg_accuracy"] = sum(accs) / len(accs) if accs else None
        else:
            s["avg_accuracy"] = None

    if not kp_stats:
        st.info("题库为空。")
        return

    df = pd.DataFrame([
        {
            "知识点": kp,
            "已练习": s["practiced"],
            "未练习": s["total"] - s["practiced"],
            "准确率": s["avg_accuracy"],
        }
        for kp, s in sorted(kp_stats.items())
    ])
    st.bar_chart(df.set_index("知识点")[["已练习", "未练习"]])
    st.dataframe(df, use_container_width=True, hide_index=True)


def render_question(q, index, total, tracker):
    """渲染单题答题界面。"""
    st.markdown(f"### 第 {index + 1} / {total} 题")
    st.caption(f"`{q['id']}` ｜ {q['exam']['type']} {q['exam']['level']} 级 {q['exam']['date']} ｜ 知识点：{', '.join(q['knowledge_points'])}")
    st.markdown(q["stem"])

    is_tf = q["type"] == "true_false"
    options = q["options"]
    keys = list(options.keys())

    # 用 session_state 控制答题状态
    state_key = f"answered_{q['id']}"
    if state_key not in st.session_state:
        st.session_state[state_key] = False

    choice = st.radio("你的选择", keys, key=f"choice_{q['id']}", disabled=st.session_state[state_key])

    if not st.session_state[state_key]:
        if st.button("提交", key=f"submit_{q['id']}"):
            is_correct = (choice == q["answer"])
            tracker.record_answer(
                q["id"], q["exam"]["type"], str(q["exam"]["level"]),
                is_correct, choice
            )
            st.session_state[state_key] = True
            st.session_state[f"last_choice_{q['id']}"] = choice
            st.rerun()
    else:
        choice = st.session_state.get(f"last_choice_{q['id']}", choice)
        is_correct = (choice == q["answer"])
        if is_correct:
            st.success(f"✅ 正确！你选了 {choice}")
        else:
            st.error(f"❌ 错误。你选了 {choice}，正确答案是 {q['answer']}")
        with st.expander("查看解析", expanded=True):
            st.markdown(q.get("explanation", "（无解析）"))

        # AI 问答
        st.markdown("---")
        st.subheader("💬 讨论此题")
        user_q = st.text_input("向 AI 提问（自动关联本题）", key=f"ai_q_{q['id']}",
                               placeholder="例如：为什么选 A？for 循环执行顺序是怎样的？")
        if st.button("提问", key=f"ai_submit_{q['id']}") and user_q:
            with st.spinner("AI 思考中..."):
                st.write_stream(chat_stream(user_q, q, choice))

"""错题本：accuracy<75% 的题复习，含按知识点/高频/随机模式 + 错题表格。"""

import pandas as pd
from pathlib import Path

from app.core.question_loader import load_all_questions, get_by_id
from app.core.practice_tracker import PracticeTracker

BANK_DIR = Path(__file__).resolve().parent.parent.parent / "question_bank"
DB_PATH = Path(__file__).resolve().parent.parent.parent / "data" / "practice_log.db"

def render_wrong_table(wrong_rows, questions):
    """错题表格：知识点列从题库查。"""
    if not wrong_rows:
        st.info("🎉 错题本为空，继续保持！")
        return
    qmap = {q["id"]: q for q in questions}
    rows = []
    for r in wrong_rows:
        q = qmap.get(r["question_id"], {})
        rows.append({
            "题目ID": r["question_id"],
            "考试": f"{r['exam_type']} {r['exam_level']}",
            "知识点": ", ".join(q.get("knowledge_points", [])),
            "题干预览": (q.get("stem", "") or "")[:30] + "...",
            "作答次数": r["attempt_count"],
            "错次": r["wrong_count"],
            "准确率": f"{r['accuracy']*100:.0f}%",
            "最近错答": r["last_wrong_answer"],
            "最近答错": r["last_wrong_at"],
        })
    st.dataframe(pd.DataFrame(rows), use_container_width=True, hide_index=True)


def render_practice(wrong_rows, questions, tracker):
    """错题练习界面。"""
    qmap = {q["id"]: q for q in questions}
    if not wrong_rows:
        return

    with st.sidebar:
        st.header("复习范围")
        mode = st.radio("出题模式", ["高频错题", "随机", "按知识点"])
        sel_kp = None
        if mode == "按知识点":
            # 从错题中收集知识点
            kps = set()
            for r in wrong_rows:
                q = qmap.get(r["question_id"])
                if q:
                    kps.update(q.get("knowledge_points", []))
            sel_kp = st.selectbox("选知识点", sorted(kps))
        if st.button("🎯 开始复习"):
            st.session_state.pop("wrong_pool", None)
            st.session_state.pop("wrong_index", None)

    if "wrong_pool" not in st.session_state:
        pool_ids = [r["question_id"] for r in wrong_rows]
        if mode == "按知识点" and sel_kp:
            pool_ids = [qid for qid in pool_ids if sel_kp in qmap.get(qid, {}).get("knowledge_points", [])]
        if mode == "随机":
            import random
            random.shuffle(pool_ids)
        # 高频错题已按 wrong_count DESC 排序
        st.session_state["wrong_pool"] = pool_ids
        st.session_state["wrong_index"] = 0

    pool = st.session_state["wrong_pool"]
    idx = st.session_state["wrong_index"]
    if not pool:
        st.info("所选范围内无错题。")
        return

    q = qmap.get(pool[idx])
    if not q:
        st.error(f"题目 {pool[idx]} 在题库中找不到")
        return

    st.markdown(f"### 复习 第 {idx+1} / {len(pool)} 题")
    st.caption(f"`{q['id']}` ｜ 知识点：{', '.join(q['knowledge_points'])}")
    st.markdown(q["stem"])

    state_key = f"wrong_answered_{q['id']}"
    if state_key not in st.session_state:
        st.session_state[state_key] = False

    options = q["options"]
    keys = list(options.keys())
    choice = st.radio("你的选择", keys, key=f"wrong_choice_{q['id']}", disabled=st.session_state[state_key])

    if not st.session_state[state_key]:
        if st.button("提交", key=f"wrong_submit_{q['id']}"):
            is_correct = (choice == q["answer"])
            tracker.record_answer(
                q["id"], q["exam"]["type"], str(q["exam"]["level"]),
                is_correct, choice
            )
            st.session_state[state_key] = True
            st.session_state[f"wrong_last_{q['id']}"] = choice
            st.rerun()
    else:
        choice = st.session_state.get(f"wrong_last_{q['id']}", choice)
        is_correct = (choice == q["answer"])
        if is_correct:
            st.success(f"✅ 正确！你选了 {choice}")
            col_a, col_b = st.columns(2)
            with col_a:
                if st.button("移出错题本", key=f"remove_{q['id']}"):
                    tracker.mark_mastered(q["id"])
                    st.success("已移出错题本。")
                    st.session_state.pop("wrong_pool", None)
                    st.rerun()
            with col_b:
                if st.button("再练一次", key=f"again_{q['id']}"):
                    st.session_state[state_key] = False
                    st.rerun()
        else:
            st.error(f"❌ 错误。你选了 {choice}，正确答案是 {q['answer']}")
        with st.expander("查看解析", expanded=True):
            st.markdown(q.get("explanation", "（无解析）"))

        if idx < len(pool) - 1:
            if st.button("下一题 ➡️", key=f"next_{q['id']}"):
                st.session_state["wrong_index"] = idx + 1
                st.rerun()


def main():
    st.set_page_config(page_title="错题本", page_icon="❌", layout="wide")
    st.title("❌ 错题本")
    st.caption("出题范围：答错过且当前准确率 < 75% 的题")

    questions = cached_questions()
    tracker = get_tracker()
    wrong_rows = tracker.get_wrong_view(order="high_freq")

    st.subheader("错题列表")
    render_wrong_table(wrong_rows, questions)

    st.markdown("---")
    st.subheader("复习")
    render_practice(wrong_rows, questions, tracker)


main()

```python
"""题库管理：表格查看 + 文本编辑 JSON + 重新加载缓存。"""
import json
import subprocess
import sys
import streamlit as st
import pandas as pd
from pathlib import Path

from app.core.question_loader import load_all_questions

BANK_DIR = Path(__file__).resolve().parent.parent.parent / "question_bank"


@st.cache_data
def cached_questions():
    return load_all_questions(BANK_DIR)


def main():
    st.set_page_config(page_title="题库管理", page_icon="📚", layout="wide")
    st.title("📚 题库管理")

    # 顶部操作
    col1, col2 = st.columns([1, 3])
    with col1:
        if st.button("🔄 重新加载题库"):
            cached_questions.clear()
            st.success("题库缓存已清空，下次读取将拿最新。")
            st.rerun()
    with col2:
        if st.button("📁 打开题库文件夹"):
            folder = str(BANK_DIR)
            if sys.platform == "win32":
                subprocess.Popen(["explorer", folder])
            elif sys.platform == "darwin":
                subprocess.Popen(["open", folder])
            else:
                subprocess.Popen(["xdg-open", folder])

    questions = cached_questions()

    # 表格查看
    st.subheader(f"题目列表（共 {len(questions)} 题）")
    if not questions:
        st.warning("题库为空。")
    else:
        df = pd.DataFrame([
            {
                "ID": q["id"],
                "考试": f"{q['exam']['type']} {q['exam']['level']} {q['exam']['date']}",
                "类型": "选择" if q["type"] == "single_choice" else "判断",
                "知识点": ", ".join(q.get("knowledge_points", [])),
                "题干预览": (q.get("stem", "") or "").replace("\n", " ")[:40],
                "答案": q.get("answer", ""),
            }
            for q in questions
        ])
        st.dataframe(df, use_container_width=True, hide_index=True)

    # 文本编辑 JSON
    st.markdown("---")
    st.subheader("📝 编辑题库文件")
    if not BANK_DIR.exists():
        st.warning(f"题库目录不存在：{BANK_DIR}")
        return
    files = sorted(BANK_DIR.glob("*.json"))
    if not files:
        st.info("题库目录下无 JSON 文件。")
        return
    sel_file = st.selectbox("选择文件", [f.name for f in files])
    file_path = BANK_DIR / sel_file

    try:
        content = file_path.read_text(encoding="utf-8")
    except Exception as e:
        st.error(f"读取失败：{e}")
        return

    edited = st.text_area("JSON 内容（直接编辑后点保存）", content, height=500, key=f"edit_{sel_file}")

    col_a, col_b = st.columns([1, 3])
    with col_a:
        if st.button("💾 保存", key=f"save_{sel_file}"):
            try:
                json.loads(edited)  # 校验
                file_path.write_text(edited, encoding="utf-8")
                st.success(f"已保存到 {sel_file}。记得点上方『重新加载题库』刷新缓存。")
            except json.JSONDecodeError as e:
                st.error(f"JSON 格式错误，未保存：{e}")
    with col_b:
        if st.button("↩️ 恢复原内容", key=f"revert_{sel_file}"):
            st.rerun()


main()
```
